   172→function timeAgo(d?: string) {
   173→  if (!d) return null;
   174→  const diff = Math.floor((Date.now() - new Date(d).getTime()) / 86400000);
   175→  if (isNaN(diff)) return null;
   176→  if (diff === 0) return 'Today';
   177→  if (diff === 1) return '1d';
   178→  if (diff < 7) return `${diff}d`;
   179→  if (diff < 31) return `${Math.floor(diff / 7)}w`;
   180→  return null;
   181→}

   614→function LatamCitiesWidget({ lang = 'EN' }: { lang?: string }) {
   615→  const [cities, setCities] = React.useState<CityScore[]>([]);
   616→  const [loading, setLoading] = React.useState(true);
   617→  const [error, setError] = React.useState(false);
   618→
   619→  const labels: Record<string, { title: string; tech: string; cost: string; safety: string; overall: string }> = {
   620→    EN: { title: 'LATAM TECH HUBS · QUALITY OF LIFE', tech: 'Tech', cost: 'Affordability', safety: 'Safety', overall: 'Score' },
   621→    ES: { title: 'HUBS TECH LATAM · CALIDAD DE VIDA',  tech: 'Tech', cost: 'Costo', safety: 'Seguridad', overall: 'Score' },
   622→    PT: { title: 'HUBS TECH LATAM · QUALIDADE DE VIDA', tech: 'Tech', cost: 'Custo', safety: 'Segurança', overall: 'Score' },
   623→  };
   624→  const lb = labels[lang] || labels.EN;
   625→
   626→  React.useEffect(() => {
   627→    let cancelled = false;
   628→    async function load() {
   629→      try {
   630→        const results = await Promise.allSettled(
   631→          LATAM_CITIES.map(c =>
   632→            fetch(`https://api.teleport.org/api/urban_areas/slug:${c.slug}/scores/`)
   633→              .then(r => r.ok ? r.json() : null)
   634→              .then(d => {
   635→                if (!d) return null;
   636→                const cats = d.categories || [];
   637→                const get = (name: string) => {
   638→                  const cat = cats.find((x: any) => x.name.toLowerCase().includes(name));
   639→                  return cat ? Math.round(cat.score_out_of_10 * 10) : 0;
   640→                };
   641→                return {
   642→                  name: c.name, flag: c.flag, slug: c.slug,
   643→                  overall: Math.round((d.teleport_city_score || 0)),
   644→                  tech: get('startup'),
   645→                  cost: get('cost'),
   646→                  safety: get('safety'),
   647→                } as CityScore;
   648→              })
   649→          )
   650→        );
   651→        if (cancelled) return;
   652→        const built = results
   653→          .filter(r => r.status === 'fulfilled' && r.value)
   654→          .map(r => (r as PromiseFulfilledResult<CityScore | null>).value!)
   655→          .sort((a, b) => b.overall - a.overall);
   656→        setCities(built);
   657→      } catch { setError(true); }
   658→      finally { if (!cancelled) setLoading(false); }
   659→    }
   660→    load();
   661→    return () => { cancelled = true; };
   662→  }, []);
   663→
   664→  if (error || (!loading && cities.length === 0)) return null;
   665→
   666→  return (
   667→    <section className="border-b border-border bg-bg px-6 md:px-10 py-8 max-w-7xl mx-auto w-full">
   668→      <div className="mono text-[9px] text-text/40 mb-4 flex items-center gap-2">
   669→        <Globe size={10} className="text-accent" /> {lb.title}
   670→      </div>
   671→      {loading ? (
   672→        <div className="flex gap-2">{[...Array(4)].map((_, i) => <div key={i} className="h-24 flex-1 bg-surface border border-border animate-pulse" />)}</div>
   673→      ) : (
   674→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   675→          {cities.slice(0, 8).map(city => (
   676→            <div key={city.slug} className="bg-surface border border-border p-3">
   677→              <div className="flex items-center justify-between mb-2">
   678→                <span className="text-[8px] font-bold text-text/80">{city.flag} {city.name}</span>
   679→                <span className="mono text-[8px] font-black text-accent">{city.overall}<span className="text-text/30">/100</span></span>
   680→              </div>
   681→              <div className="space-y-1">
   682→                {[
   683→                  { label: lb.tech, val: city.tech },
   684→                  { label: lb.cost, val: city.cost },
   685→                  { label: lb.safety, val: city.safety },
   686→                ].map(row => (
   687→                  <div key={row.label} className="flex items-center gap-1">
   688→                    <span className="mono text-[7px] text-text/30 w-16 shrink-0">{row.label}</span>
   689→                    <div className="flex-1 h-1 bg-border rounded-full overflow-hidden">
   690→                      <div className="h-full bg-accent/60 rounded-full" style={{ width: `${row.val}%` }} />
   691→                    </div>
   692→                    <span className="mono text-[7px] text-text/50 w-5 text-right">{row.val}</span>
   693→                  </div>
   694→                ))}
   695→              </div>
   696→            </div>
   697→          ))}
   698→        </div>
   699→      )}
   700→    </section>
   701→  );
